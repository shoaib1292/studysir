    49→function previewText(connection: ConnectionDTO, meId: string): { text: string; italic: boolean } {
    50→  const lm = connection.lastMessage
    51→  if (!lm) return { text: 'New connection request — say hi!', italic: false }
    52→  const involvesMe = lm.senderId === meId
    53→  const isParty = lm.senderId === connection.teacher.id || lm.senderId === connection.student.id
    54→  if (!isParty) return { text: lm.content, italic: true } // system message
    55→  return { text: involvesMe ? `You: ${lm.content}` : lm.content, italic: false }
    56→}

   204→async function copyText(text: string): Promise<boolean> {
   205→  try {
   206→    if (typeof navigator !== 'undefined' && navigator.clipboard) {
   207→      await navigator.clipboard.writeText(text)
   208→      return true
   209→    }
   210→  } catch {
   211→    // fall through to legacy path (insecure context)
   212→  }
   213→  try {
   214→    const ta = document.createElement('textarea')
   215→    ta.value = text
   216→    ta.style.position = 'fixed'
   217→    ta.style.opacity = '0'
   218→    document.body.appendChild(ta)
   219→    ta.select()
   220→    const ok = document.execCommand('copy')
   221→    document.body.removeChild(ta)
   222→    return ok
   223→  } catch {
   224→    return false
   225→  }
   226→}

   345→function ChatThread({
   346→  connectionId,
   347→  onListChanged,
   348→}: {
   349→  connectionId: string
   350→  onListChanged: () => void
   351→}) {
   352→  const me = useAppStore((s) => s.me)!
   353→  const back = useAppStore((s) => s.back)
   354→  const refreshMe = useAppStore((s) => s.refreshMe)
   355→
   356→  const [data, setData] = useState<{ connection: ConnectionDTO; messages: MessageDTO[] } | null>(null)
   357→  const [failed, setFailed] = useState(false)
   358→  const [draft, setDraft] = useState('')
   359→  const [sending, setSending] = useState(false)
   360→  const [busy, setBusy] = useState(false)
   361→  const [dialog, setDialog] = useState<ThreadDialog>(null)
   362→  const [reportReason, setReportReason] = useState('')
   363→  const [typingName, setTypingName] = useState<string | null>(null)
   364→  const [showJump, setShowJump] = useState(false)
   365→  /** snapshot of unseen messages when the thread was first opened (drives the "new" divider) */
   366→  const [unreadInfo, setUnreadInfo] = useState<{ count: number; firstId: string | null } | null>(null)
   367→  const unreadCaptured = useRef(false)
   368→  const typingTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   369→  const lastTypingSent = useRef(0)
   370→  const scrollRef = useRef<HTMLDivElement>(null)
   371→  const fileInputRef = useRef<HTMLInputElement>(null)
   372→  const inputRef = useRef<HTMLInputElement>(null)
   373→  const onlineIds = useAppStore((s) => s.onlineIds)
   374→
   375→  const load = useCallback(async () => {
   376→    try {
   377→      const d = await api.getConnection(connectionId)
   378→      setData(d)
   379→      if (!unreadCaptured.current) {
   380→        unreadCaptured.current = true
   381→        setUnreadInfo(d.unread)
   382→      }
   383→      setFailed(false)
   384→    } catch {
   385→      setFailed(true)
   386→    }
   387→  }, [connectionId])
   388→
   389→  // New thread → capture a fresh unread snapshot
   390→  useEffect(() => {
   391→    unreadCaptured.current = false
   392→    setUnreadInfo(null)
   393→  }, [connectionId])
   394→
   395→  useEffect(() => {
   396→    void load()
   397→  }, [load])
   398→
   399→  // Realtime: live messages, status changes, read receipts + typing indicator.
   400→  // A slow interval poll stays as a safety net for missed events.
   401→  useEffect(() => {
   402→    // ensure socket identity, then subscribe
   403→    getSocket(me.id, me.name)
   404→
   405→    const onMessage = (p: { connectionId?: string; message?: MessageDTO }) => {
   406→      if (!p || p.connectionId !== connectionId) return
   407→      if (p.message && p.message.senderId === me.id) return // own messages handled by send flow
   408→      void load()
   409→    }
   410→    const onUpdated = (p: { connectionId?: string }) => {
   411→      if (!p || p.connectionId !== connectionId) return
   412→      void load()
   413→      onListChanged()
   414→    }
   415→    const onRead = (p: { connectionId?: string }) => {
   416→      if (!p || p.connectionId !== connectionId) return
   417→      setData((d) =>
   418→        d
   419→          ? {
   420→              ...d,
   421→              messages: d.messages.map((m) =>
   422→                m.senderId === me.id && !m.readAt ? { ...m, readAt: new Date().toISOString() } : m
   423→              ),
   424→            }
   425→          : d
   426→      )
   427→    }
   428→    const onTyping = (p: { connectionId?: string; userId?: string; name?: string; isTyping?: boolean }) => {
   429→      if (!p || p.connectionId !== connectionId || p.userId === me.id) return
   430→      if (typingTimer.current) clearTimeout(typingTimer.current)
   431→      if (p.isTyping) {
   432→        setTypingName(p.name || 'Someone')
   433→        typingTimer.current = setTimeout(() => setTypingName(null), 3000)
   434→      } else {
   435→        setTypingName(null)
   436→      }
   437→    }
   438→
   439→    onEvent(RT.chatMessage, onMessage)
   440→    onEvent(RT.chatUpdated, onUpdated)
   441→    onEvent(RT.chatRead, onRead)
   442→    onEvent('typing', onTyping)
   443→
   444→    const timer = setInterval(() => void load(), 15000)
   445→    return () => {
   446→      clearInterval(timer)
   447→      offEvent(RT.chatMessage, onMessage)
   448→      offEvent(RT.chatUpdated, onUpdated)
   449→      offEvent(RT.chatRead, onRead)
   450→      offEvent('typing', onTyping)
   451→      if (typingTimer.current) clearTimeout(typingTimer.current)
   452→    }
   453→  }, [connectionId, load, onListChanged, me.id, me.name])
   454→
   455→  // Auto-scroll to the newest message (also when typing indicator appears)
   456→  const messageCount = data?.messages.length ?? 0
   457→  useEffect(() => {
   458→    const el = scrollRef.current
   459→    if (el) el.scrollTop = el.scrollHeight
   460→  }, [messageCount, connectionId, typingName])
   461→
   462→  function handleScroll() {
   463→    const el = scrollRef.current
   464→    if (!el) return
   465→    const distance = el.scrollHeight - el.scrollTop - el.clientHeight
   466→    setShowJump(distance > 240)
   467→  }
   468→
   469→  function jumpToLatest() {
   470→    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: 'smooth' })
   471→  }
   472→
   473→  const connection = data?.connection
   474→  const other = connection ? otherParty(connection, me.id) : null
   475→  const otherOnline = !!other && onlineIds.includes(other.id)
   476→
   477→  const blocked = !!connection?.blockedBy
   478→  const iBlocked = connection?.blockedBy === me.id
   479→  const decided = !!connection && ['HIRED', 'REJECTED', 'EXPIRED'].includes(connection.status)
   480→  const isPostOwner = connection?.student.id === me.id
   481→  const showActions = !!connection && !blocked && !decided
   482→
   483→  function handleDraftChange(value: string) {
   484→    setDraft(value)
   485→    if (!connection || !other) return
   486→    const now = Date.now()
   487→    if (value.trim() && now - lastTypingSent.current > 1500) {
   488→      lastTypingSent.current = now
   489→      emitTyping(other.id, connectionId, true)
   490→    } else if (!value.trim() && lastTypingSent.current) {
   491→      lastTypingSent.current = 0
   492→      emitTyping(other.id, connectionId, false)
   493→    }
   494→  }
   495→
   496→  async function send(e: FormEvent) {
   497→    e.preventDefault()
   498→    const content = draft.trim()
   499→    if (!content || !connection || sending) return
   500→    setSending(true)
   501→    setDraft('')
   502→    const optimistic: MessageDTO = {
   503→      id: `tmp-${Date.now()}`,
   504→      connectionId: connection.id,
   505→      senderId: me.id,
   506→      sender: { id: me.id, name: me.name, avatar: me.avatar },
   507→      content,
   508→      image: null,
   509→      system: false,
   510→      createdAt: new Date().toISOString(),
   511→      readAt: null,
   512→    }
   513→    setData((d) => (d ? { ...d, messages: [...d.messages, optimistic] } : d))
   514→    if (connection && other) emitTyping(other.id, connection.id, false)
   515→    try {
   516→      await api.sendMessage(connection.id, { content })
   517→      await load()
   518→    } catch (err) {
   519→      setData((d) => (d ? { ...d, messages: d.messages.filter((m) => m.id !== optimistic.id) } : d))
   520→      setDraft(content)
   521→      toast.error('Message not sent', { description: errorMessage(err) })
   522→    } finally {
   523→      setSending(false)
   524→    }
   525→  }

   527→  async function sendImage(file: File) {
   528→    if (!connection || sending) return
   529→    setSending(true)
   530→    const optimisticId = `tmp-${Date.now()}`
