    43→    43→function previewText(connection: ConnectionDTO, meId: string): { text: string; italic: boolean } {
    44→    44→  const lm = connection.lastMessage
    45→    45→  if (!lm) return { text: 'New connection request — say hi!', italic: false }
    46→    46→  const involvesMe = lm.senderId === meId
    47→    47→  const isParty = lm.senderId === connection.teacher.id || lm.senderId === connection.student.id
    48→    48→  if (!isParty) return { text: lm.content, italic: true } // system message
    49→    49→  return { text: involvesMe ? `You: ${lm.content}` : lm.content, italic: false }
    50→    50→}

   191→   191→function ChatThread({
   192→   192→  connectionId,
   193→   193→  onListChanged,
   194→   194→}: {
   195→   195→  connectionId: string
   196→   196→  onListChanged: () => void
   197→   197→}) {
   198→   198→  const me = useAppStore((s) => s.me)!
   199→   199→  const back = useAppStore((s) => s.back)
   200→   200→  const refreshMe = useAppStore((s) => s.refreshMe)
   201→   201→
   202→   202→  const [data, setData] = useState<{ connection: ConnectionDTO; messages: MessageDTO[] } | null>(null)
   203→   203→  const [failed, setFailed] = useState(false)
   204→   204→  const [draft, setDraft] = useState('')
   205→   205→  const [sending, setSending] = useState(false)
   206→   206→  const [busy, setBusy] = useState(false)
   207→   207→  const [dialog, setDialog] = useState<ThreadDialog>(null)
   208→   208→  const [reportReason, setReportReason] = useState('')
   209→   209→  const [typingName, setTypingName] = useState<string | null>(null)
   210→   210→  const [showJump, setShowJump] = useState(false)
   211→   211→  const typingTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   212→   212→  const lastTypingSent = useRef(0)
   213→   213→  const scrollRef = useRef<HTMLDivElement>(null)
   214→   214→  const onlineIds = useAppStore((s) => s.onlineIds)
   215→   215→
   216→   216→  const load = useCallback(async () => {
   217→   217→    try {
   218→   218→      const d = await api.getConnection(connectionId)
   219→   219→      setData(d)
   220→   220→      setFailed(false)
   221→   221→    } catch {
   222→   222→      setFailed(true)
   223→   223→    }
   224→   224→  }, [connectionId])
   225→   225→
   226→   226→  useEffect(() => {
   227→   227→    void load()
   228→   228→  }, [load])
   229→   229→
   230→   230→  // Realtime: live messages, status changes, read receipts + typing indicator.
   231→   231→  // A slow interval poll stays as a safety net for missed events.
   232→   232→  useEffect(() => {
   233→   233→    // ensure socket identity, then subscribe
   234→   234→    getSocket(me.id, me.name)
   235→   235→
   236→   236→    const onMessage = (p: { connectionId?: string; message?: MessageDTO }) => {
   237→   237→      if (!p || p.connectionId !== connectionId) return
   238→   238→      if (p.message && p.message.senderId === me.id) return // own messages handled by send flow
   239→   239→      void load()
   240→   240→    }
   241→   241→    const onUpdated = (p: { connectionId?: string }) => {
   242→   242→      if (!p || p.connectionId !== connectionId) return
   243→   243→      void load()
   244→   244→      onListChanged()
   245→   245→    }
   246→   246→    const onRead = (p: { connectionId?: string }) => {
   247→   247→      if (!p || p.connectionId !== connectionId) return
   248→   248→      setData((d) =>
   249→   249→        d
   250→   250→          ? {
   251→   251→              ...d,
   252→   252→              messages: d.messages.map((m) =>
   253→   253→                m.senderId === me.id && !m.readAt ? { ...m, readAt: new Date().toISOString() } : m
   254→   254→              ),
   255→   255→            }
   256→   256→          : d
   257→   257→      )
   258→   258→    }
   259→   259→    const onTyping = (p: { connectionId?: string; userId?: string; name?: string; isTyping?: boolean }) => {
   260→   260→      if (!p || p.connectionId !== connectionId || p.userId === me.id) return
   261→   261→      if (typingTimer.current) clearTimeout(typingTimer.current)
   262→   262→      if (p.isTyping) {
   263→   263→        setTypingName(p.name || 'Someone')
   264→   264→        typingTimer.current = setTimeout(() => setTypingName(null), 3000)
   265→   265→      } else {
   266→   266→        setTypingName(null)
   267→   267→      }
   268→   268→    }
   269→   269→
   270→   270→    onEvent(RT.chatMessage, onMessage)
   271→   271→    onEvent(RT.chatUpdated, onUpdated)
   272→   272→    onEvent(RT.chatRead, onRead)
   273→   273→    onEvent('typing', onTyping)
   274→   274→
   275→   275→    const timer = setInterval(() => void load(), 15000)
   276→   276→    return () => {
   277→   277→      clearInterval(timer)
   278→   278→      offEvent(RT.chatMessage, onMessage)
   279→   279→      offEvent(RT.chatUpdated, onUpdated)
   280→   280→      offEvent(RT.chatRead, onRead)
   281→   281→      offEvent('typing', onTyping)
   282→   282→      if (typingTimer.current) clearTimeout(typingTimer.current)
   283→   283→    }
   284→   284→  }, [connectionId, load, onListChanged, me.id, me.name])
   285→   285→
   286→   286→  // Auto-scroll to the newest message (also when typing indicator appears)
   287→   287→  const messageCount = data?.messages.length ?? 0
   288→   288→  useEffect(() => {
   289→   289→    const el = scrollRef.current
   290→   290→    if (el) el.scrollTop = el.scrollHeight
   291→   291→  }, [messageCount, connectionId, typingName])
   292→   292→
   293→   293→  function handleScroll() {
   294→   294→    const el = scrollRef.current
   295→   295→    if (!el) return
   296→   296→    const distance = el.scrollHeight - el.scrollTop - el.clientHeight
   297→   297→    setShowJump(distance > 240)
   298→   298→  }
   299→   299→
   300→   300→  function jumpToLatest() {
   301→   301→    scrollRef.current?.scrollTo({ top: scrollRef.current.scrollHeight, behavior: 'smooth' })
   302→   302→  }
   303→   303→
   304→   304→  const connection = data?.connection
   305→   305→  const other = connection ? otherParty(connection, me.id) : null
   306→   306→  const otherOnline = !!other && onlineIds.includes(other.id)
   307→   307→
   308→   308→  const blocked = !!connection?.blockedBy
   309→   309→  const iBlocked = connection?.blockedBy === me.id
   310→   310→  const decided = !!connection && ['HIRED', 'REJECTED', 'EXPIRED'].includes(connection.status)
   311→   311→  const isPostOwner = connection?.student.id === me.id ?? false
   312→   312→  const showActions = !!connection && !blocked && !decided
   313→   313→
   314→   314→  function handleDraftChange(value: string) {
   315→   315→    setDraft(value)
   316→   316→    if (!connection || !other) return
   317→   317→    const now = Date.now()
   318→   318→    if (value.trim() && now - lastTypingSent.current > 1500) {
   319→   319→      lastTypingSent.current = now
   320→   320→      emitTyping(other.id, connectionId, true)
   321→   321→    } else if (!value.trim() && lastTypingSent.current) {
   322→   322→      lastTypingSent.current = 0
   323→   323→      emitTyping(other.id, connectionId, false)
   324→   324→    }
   325→   325→  }
   326→   326→
   327→   327→  async function send(e: FormEvent) {
   328→   328→    e.preventDefault()
   329→   329→    const content = draft.trim()
   330→   330→    if (!content || !connection || sending) return
   331→   331→    setSending(true)
   332→   332→    setDraft('')
   333→   333→    const optimistic: MessageDTO = {
   334→   334→      id: `tmp-${Date.now()}`,
   335→   335→      connectionId: connection.id,
   336→   336→      senderId: me.id,
   337→   337→      sender: { id: me.id, name: me.name, avatar: me.avatar },
   338→   338→      content,
   339→   339→      system: false,
   340→   340→      createdAt: new Date().toISOString(),
   341→   341→    }
   342→   342→    setData((d) => (d ? { ...d, messages: [...d.messages, optimistic] } : d))
   343→   343→    if (connection && other) emitTyping(other.id, connection.id, false)
   344→   344→    try {
   345→   345→      await api.sendMessage(connection.id, content)
   346→   346→      await load()
   347→   347→    } catch (err) {
   348→   348→      setData((d) => (d ? { ...d, messages: d.messages.filter((m) => m.id !== optimistic.id) } : d))
   349→   349→      setDraft(content)
   350→   350→      toast.error('Message not sent', { description: errorMessage(err) })
   351→   351→    } finally {
   352→   352→      setSending(false)
   353→   353→    }
   354→   354→  }
   355→   355→
   356→   356→  async function decide(action: 'HIRE' | 'REJECT' | 'BLOCK' | 'REPORT', reason?: string) {
   357→   357→    if (!connection || busy) return
   358→   358→    setBusy(true)
   359→   359→    try {
   360→   360→      const res = await api.decide(connection.id, action, reason)
   361→   361→      setData((d) => (d ? { ...d, connection: res.connection } : d))
   362→   362→      await refreshMe()
   363→   363→      if (action === 'HIRE') {
   364→   364→        toast.success('Teacher hired! 🎉', { description: 'Conversation is now locked.' })
   365→   365→      } else if (action === 'REJECT') {
   366→   366→        toast.success(
   367→   367→          'Request rejected',
   368→   368→          res.connection.refunded
   369→   369→            ? { description: `${res.connection.coinsSpent} coins refunded to the teacher.` }
   370→   370→            : { description: 'No coins were refunded — chat had already started.' }
   371→   371→        )
   372→   372→      } else if (action === 'BLOCK') {
   373→   373→        toast.success('User blocked')
   374→   374→      } else if (action === 'REPORT') {
   375→   375→        toast.success('Report submitted', { description: 'Our team will review it. Thank you.' })
   376→   376→      }
   377→   377→      setDialog(null)
   378→   378→      setReportReason('')
   379→   379→      onListChanged()
   380→   380→    } catch (err) {
   381→   381→      toast.error('Action failed', { description: errorMessage(err) })
   382→   382→    } finally {
   383→   383→      setBusy(false)
   384→   384→    }
   385→   385→  }
   386→   386→
   387→   387→  async function unblock() {
   388→   388→    if (!connection || busy) return
   389→   389→    setBusy(true)
   390→   390→    try {
   391→   391→      await api.decide(connection.id, 'UNBLOCK')
   392→   392→      toast.success('User unblocked', { description: 'You can chat again.' })
   393→   393→      await load()
   394→   394→      onListChanged()
   395→   395→    } catch (err) {
   396→   396→      toast.error('Could not unblock', { description: errorMessage(err) })
   397→   397→    } finally {
   398→   398→      setBusy(false)
   399→   399→    }
   400→   400→  }
   401→   401→
   402→   402→  // Build message nodes with system rows + day chips
   403→   403→  const messageNodes: ReactNode[] = []
   404→   404→  if (data) {
   405→   405→    let prevDay = ''
   406→   406→    for (const m of data.messages) {
   407→   407→      const day = dayLabel(m.createdAt)
   408→   408→      if (day !== prevDay) {
   409→   409→        messageNodes.push(<DayChip key={`day-${m.id}`} label={day} />)
   410→   410→        prevDay = day
   411→   411→      }
   412→   412→      if (m.system) messageNodes.push(<SystemMessage key={m.id} content={m.content} />)
   413→   413→      else messageNodes.push(<MessageBubble key={m.id} message={m} mine={m.senderId === me.id} />)
   414→   414→    }
   415→   415→  }
   416→   416→
   417→   417→  if (failed) {
   418→   418→    return (
   419→   419→      <div className="flex h-full flex-col items-center justify-center gap-2 bg-muted p-6 text-center">
   420→   420→        <EmptyState icon={MessageCircle} title="Chat not found" hint="It may have been removed." />
   421→   421→      </div>
   422→   422→    )
   423→   423→  }
   424→   424→
   425→   425→  return (
   426→   426→    <div className="flex h-full min-w-0 flex-col">
   427→   427→      {/* Blue header */}
   428→   428→      <div className="flex items-center gap-3 bg-gradient-to-r from-[#1877F2] to-[#0C63D8] p-3 text-white">
   429→   429→        <button
   430→   430→          type="button"
   431→   431→          onClick={back}
   432→   432→          aria-label="Back to chats"
   433→   433→          className="rounded-full p-1.5 transition-colors hover:bg-white/15 lg:hidden"
   434→   434→        >
   435→   435→          <ArrowLeft className="size-5" />
   436→   436→        </button>
   437→   437→        {other ? (
   438→   438→          <>
   439→   439→            <UserAvatar src={other.avatar} name={other.name} className="size-10 ring-2 ring-white/40" />
   440→   440→            <div className="min-w-0 flex-1">
   441→   441→              <p className="truncate font-bold leading-tight">{other.name}</p>
   442→   442→              <p className="truncate text-[11px] opacity-90">
   443→   443→                {typingName ? (
   444→   444→                  <span className="italic">typing…</span>
   445→   445→                ) : otherOnline ? (
   446→   446→                  <span className="flex items-center gap-1">
   447→   447→                    <span className="inline-block size-1.5 animate-pulse rounded-full bg-green-400" />
   448→   448→                    Active now

   753→   753→          </div>
