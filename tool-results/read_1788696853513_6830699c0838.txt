   108→function CopyButton({ value, label }: { value: string; label: string }) {
   109→  const [copied, setCopied] = useState(false)
   110→
   111→  async function copy() {
   112→    try {
   113→      await navigator.clipboard.writeText(value)
   114→      setCopied(true)
   115→      toast.success(`${label} copied`)
   116→      window.setTimeout(() => setCopied(false), 1600)
   117→    } catch {
   118→      toast.error('Could not copy', { description: 'Clipboard is unavailable in this browser.' })
   119→    }
   120→  }
   121→
   122→  return (
   123→    <button
   124→      type="button"
   125→      aria-label={`Copy ${label.toLowerCase()}`}
   126→      title={`Copy ${label.toLowerCase()}`}
   127→      onClick={() => void copy()}
   128→      className="inline-flex size-6 shrink-0 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-secondary hover:text-foreground"
   129→    >
   130→      {copied ? <Check className="size-3.5 text-green-600 dark:text-green-400" /> : <Copy className="size-3.5" />}
   131→    </button>
   132→  )
   133→}

   135→/** Outcome caption shown on resolved top-ups (per product spec). */
   136→function outcomeOf(topup: TopUpDTO): string | null {
   137→  if (topup.status === 'APPROVED') {
   138→    return topup.kind === 'TEACHER_COINS' ? 'Coins were credited instantly.' : 'Money added to wallet.'
   139→  }
   140→  if (topup.status === 'REJECTED' && topup.kind === 'TEACHER_COINS') {
   141→    return 'Coins were clawed back.'
   142→  }
   143→  return null
   144→}

   597→                value={note}
   598→                onChange={(e) => setNote(e.target.value)}
   599→                placeholder="Admin note (optional) — saved with the request"

   625→export default function PaymentsTab() {
   626→  const [tab, setTab] = useState<'proofs' | 'withdrawals'>('proofs')
   627→
   628→  // Payment proofs (top-up queue)
   629→  const [topFilter, setTopFilter] = useState<PayFilter>('PENDING')
   630→  const [topups, setTopups] = useState<TopUpDTO[] | null>(null)
   631→
   632→  // Withdrawals queue
   633→  const [wdFilter, setWdFilter] = useState<PayFilter>('PENDING')
   634→  const [withdrawals, setWithdrawals] = useState<WithdrawalDTO[] | null>(null)
   635→
   636→  const loadTopups = useCallback(async () => {
   637→    try {
   638→      const d = await api.adminTopUps(topFilter === 'ALL' ? undefined : topFilter)
   639→      setTopups(d.topups)
   640→    } catch (e) {
   641→      setTopups([])
   642→      toast.error('Could not load payment proofs', { description: errorMessage(e) })
   643→    }
   644→  }, [topFilter])
   645→
   646→  const loadWithdrawals = useCallback(async () => {
   647→    try {
   648→      const d = await api.adminWithdrawals(wdFilter === 'ALL' ? undefined : wdFilter)
   649→      setWithdrawals(d.withdrawals)
   650→    } catch (e) {
   651→      setWithdrawals([])
   652→      toast.error('Could not load withdrawals', { description: errorMessage(e) })
   653→    }
   654→  }, [wdFilter])
   655→
   656→  useEffect(() => {
   657→    // eslint-disable-next-line react-hooks/set-state-in-effect
   658→    void loadTopups()
   659→  }, [loadTopups])
   660→
   661→  useEffect(() => {
   662→    if (tab !== 'withdrawals') return
   663→    // eslint-disable-next-line react-hooks/set-state-in-effect
   664→    void loadWithdrawals()
   665→  }, [tab, loadWithdrawals])
   666→
   667→  return (
   668→    <div className="space-y-3">
   669→      <Tabs value={tab} onValueChange={(v) => setTab(v as 'proofs' | 'withdrawals')}>
   670→        <TabsList className="w-full sm:w-auto">
   671→          <TabsTrigger value="proofs" className="gap-1.5">
   672→            <Wallet className="size-4" />
   673→            Payment Proofs
   674→          </TabsTrigger>
   675→          <TabsTrigger value="withdrawals" className="gap-1.5">
   676→            <Landmark className="size-4" />
   677→            Withdrawals
   678→          </TabsTrigger>
   679→        </TabsList>
   680→
   681→        {/* ===== Payment proofs (top-up requests) ===== */}
   682→        <TabsContent value="proofs" className="mt-3 space-y-3">
   683→          <div className="flex flex-wrap items-center gap-2">
   684→            <FilterPills value={topFilter} onChange={setTopFilter} />
   685→            <Button size="sm" variant="ghost" className="ml-auto gap-1.5" onClick={() => void loadTopups()}>
   686→              <RefreshCw className="size-3.5" />
   687→              Refresh
   688→            </Button>
   689→          </div>
   690→
   691→          {topups === null ? (
   692→            <div className="space-y-3">
   693→              {[0, 1, 2].map((i) => (
   694→                <Skeleton key={i} className="h-40 rounded-xl" />
   695→              ))}
   696→            </div>
   697→          ) : topups.length === 0 ? (
   698→            <EmptyState
   699→              icon={Wallet}
   700→              title={topFilter === 'PENDING' ? 'Queue is clear' : 'Nothing here'}
   701→              hint={
   702→                topFilter === 'PENDING'
   703→                  ? 'No payment screenshots waiting for review — nice work!'
   704→                  : 'No payment proofs match this filter.'
   705→              }
   706→            />
   707→          ) : (
   708→            <div className="space-y-3">
   709→              {topups.map((t) => (
   710→                <TopUpCard key={t.id} topup={t} onChanged={() => void loadTopups()} />
   711→              ))}
   712→            </div>
   713→          )}
   714→        </TabsContent>
   715→
   716→        {/* ===== Withdrawals ===== */}
   717→        <TabsContent value="withdrawals" className="mt-3 space-y-3">
   718→          <div className="flex flex-wrap items-center gap-2">
   719→            <FilterPills value={wdFilter} onChange={setWdFilter} />
   720→            <Button size="sm" variant="ghost" className="ml-auto gap-1.5" onClick={() => void loadWithdrawals()}>
   721→              <RefreshCw className="size-3.5" />
   722→              Refresh
   723→            </Button>
   724→          </div>
   725→
   726→          {withdrawals === null ? (
   727→            <div className="space-y-3">
   728→              {[0, 1, 2].map((i) => (
   729→                <Skeleton key={i} className="h-44 rounded-xl" />
   730→              ))}
   731→            </div>
   732→          ) : withdrawals.length === 0 ? (
   733→            <EmptyState
   734→              icon={Landmark}
   735→              title={wdFilter === 'PENDING' ? 'No pending withdrawals' : 'Nothing here'}
   736→              hint={
   737→                wdFilter === 'PENDING'
   738→                  ? 'No withdrawal requests waiting for payout.'
   739→                  : 'No withdrawals match this filter.'
   740→              }
   741→            />
   742→          ) : (
   743→            <div className="space-y-3">
   744→              {withdrawals.map((w) => (
   745→                <WithdrawalCard key={w.id} withdrawal={w} onChanged={() => void loadWithdrawals()} />
   746→              ))}
   747→            </div>
   748→          )}
   749→        </TabsContent>
   750→      </Tabs>
   751→    </div>
   752→  )
