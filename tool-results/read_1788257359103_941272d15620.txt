    50→function previewText(connection: ConnectionDTO, meId: string): { text: string; italic: boolean } {
    51→  const lm = connection.lastMessage
    52→  if (!lm) return { text: 'New connection request — say hi!', italic: false }
    53→  const involvesMe = lm.senderId === meId
    54→  const isParty = lm.senderId === connection.teacher.id || lm.senderId === connection.student.id
    55→  if (!isParty) return { text: lm.content, italic: true } // system message
    56→  return { text: involvesMe ? `You: ${lm.content}` : lm.content, italic: false }
    57→}

   212→async function copyText(text: string): Promise<boolean> {
   213→  try {
   214→    if (typeof navigator !== 'undefined' && navigator.clipboard) {
   215→      await navigator.clipboard.writeText(text)
   216→      return true
   217→    }
   218→  } catch {
   219→    // fall through to legacy path (insecure context)
   220→  }
   221→  try {
   222→    const ta = document.createElement('textarea')
   223→    ta.value = text
   224→    ta.style.position = 'fixed'
   225→    ta.style.opacity = '0'
   226→    document.body.appendChild(ta)
   227→    ta.select()
   228→    const ok = document.execCommand('copy')
   229→    document.body.removeChild(ta)
   230→    return ok
   231→  } catch {
   232→    return false
   233→  }
   234→}

   451→function ChatThread({
   452→  connectionId,
   453→  onListChanged,
   454→}: {
   455→  connectionId: string
   456→  onListChanged: () => void
   457→}) {
   458→  const me = useAppStore((s) => s.me)!
   459→  const back = useAppStore((s) => s.back)
   460→  const refreshMe = useAppStore((s) => s.refreshMe)
   461→
   462→  const [data, setData] = useState<{ connection: ConnectionDTO; messages: MessageDTO[] } | null>(null)
   463→  const [failed, setFailed] = useState(false)
   464→  const [draft, setDraft] = useState('')
   465→  const [sending, setSending] = useState(false)
   466→  const [busy, setBusy] = useState(false)
   467→  const [dialog, setDialog] = useState<ThreadDialog>(null)
   468→  const [reportReason, setReportReason] = useState('')
   469→  const [typingName, setTypingName] = useState<string | null>(null)
   470→  const [showJump, setShowJump] = useState(false)
   471→  /** snapshot of unseen messages when the thread was first opened (drives the "new" divider) */
   472→  const [unreadInfo, setUnreadInfo] = useState<{ count: number; firstId: string | null } | null>(null)
   473→  const unreadCaptured = useRef(false)
   474→  const typingTimer = useRef<ReturnType<typeof setTimeout> | null>(null)
   475→  const lastTypingSent = useRef(0)
   476→  const scrollRef = useRef<HTMLDivElement>(null)
   477→  const fileInputRef = useRef<HTMLInputElement>(null)
   478→  const inputRef = useRef<HTMLInputElement>(null)
   479→  const onlineIds = useAppStore((s) => s.onlineIds)
   480→
   481→  const load = useCallback(async () => {
   482→    try {
   483→      const d = await api.getConnection(connectionId)
   484→      setData(d)
   485→      if (!unreadCaptured.current) {
   486→        unreadCaptured.current = true
   487→        setUnreadInfo(d.unread)
   488→      }
   489→      setFailed(false)
   490→    } catch {
   491→      setFailed(true)
   492→    }
   493→  }, [connectionId])
   494→
   495→  // New thread → capture a fresh unread snapshot
   496→  useEffect(() => {
   497→    unreadCaptured.current = false
   498→    setUnreadInfo(null)
   499→  }, [connectionId])
   500→
   501→  useEffect(() => {
   502→    void load()
   503→  }, [load])
   504→
   505→  // Realtime: live messages, status changes, read receipts + typing indicator.
   506→  // A slow interval poll stays as a safety net for missed events.
   507→  useEffect(() => {
   508→    // ensure socket identity, then subscribe
   509→    getSocket(me.id, me.name)
   510→
   511→    const onMessage = (p: { connectionId?: string; message?: MessageDTO }) => {
   512→      if (!p || p.connectionId !== connectionId) return
   513→      if (p.message && p.message.senderId === me.id) return // own messages handled by send flow
   514→      void load()
   515→    }
   516→    const onUpdated = (p: { connectionId?: string }) => {
   517→      if (!p || p.connectionId !== connectionId) return
   518→      void load()
