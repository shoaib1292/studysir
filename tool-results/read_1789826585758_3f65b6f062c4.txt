    52→    52→function personaOf(raw: string | null): Partial<AiPersona> {
    53→    53→  if (raw) {
    54→    54→    try {
    55→    55→      const parsed = JSON.parse(raw) as Partial<AiPersona>
    56→    56→      if (parsed && typeof parsed === 'object') return parsed
    57→    57→    } catch {
    58→    58→      // broken JSON → empty object → defaults below
    59→    59→    }
    60→    60→  }
    61→    61→  return {}
    62→    62→}

    64→    64→function clampNum(v: string, fallback: number, min: number, max: number): number {
    65→    65→  const n = Number(v)
    66→    66→  if (!Number.isFinite(n)) return fallback
    67→    67→  return Math.min(max, Math.max(min, n))
    68→    68→}
