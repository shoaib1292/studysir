    51→/** Safe aiPersona JSON parse — never throws; missing fields fall back at call sites. */
    52→function personaOf(raw: string | null): Partial<AiPersona> {
    53→  if (raw) {
    54→    try {
    55→      const parsed = JSON.parse(raw) as Partial<AiPersona>
    56→      if (parsed && typeof parsed === 'object') return parsed
    57→    } catch {
    58→      // broken JSON → empty object → defaults below
    59→    }
    60→  }
    61→  return {}
    62→}
    63→
    64→function clampNum(v: string, fallback: number, min: number, max: number): number {
    65→  const n = Number(v)
    66→  if (!Number.isFinite(n)) return fallback
    67→  return Math.min(max, Math.max(min, n))
    68→}
    69→
    70→function fmtHour(h: number): string {
    71→  return `${Math.min(24, Math.max(0, Math.round(h)))}:00`

   189→interface AgentFormState {
   190→  name: string
   191→  role: 'TEACHER' | 'STUDENT'
   192→  avatar: string | null
   193→  headline: string
   194→  bio: string
   195→  city: string
   196→  subjects: string
   197→  feeMin: string
   198→  feeMax: string
   199→  coins: string

   201→  style: string
   202→  activeFrom: string
   203→  activeTo: string
   204→  minDelaySec: string
   205→  maxDelaySec: string
   206→  mergeWindowSec: string
   207→  replyChance: string
   208→  declineLines: string
   209→}
   210→
   211→function buildForm(agent: AiAgentDTO | null): AgentFormState {
   212→  if (!agent) {
   213→    return {
   214→      name: '',
   215→      role: 'TEACHER',
   216→      avatar: null,
   217→      headline: '',
   218→      bio: '',
   219→      city: '',
   220→      subjects: '',
   221→      feeMin: '',
   222→      feeMax: '',
   223→      coins: '100',
   224→      grantCoins: '',
   225→      style: '',
   226→      activeFrom: '5',
   227→      activeTo: '17',
   228→      minDelaySec: '20',
   229→      maxDelaySec: '90',
   230→      mergeWindowSec: '7',
   231→      replyChance: '0.9',
   232→      declineLines: '',
   233→    }
   234→  }
   235→  const p = personaOf(agent.aiPersona)
   236→  return {
   237→    name: agent.name,
   238→    role: agent.role === 'STUDENT' ? 'STUDENT' : 'TEACHER',
   239→    avatar: agent.avatar,
   240→    headline: agent.headline ?? '',
   241→    bio: agent.bio ?? '',
   242→    city: agent.city ?? '',
   243→    subjects: agent.subjects ?? '',
   244→    feeMin: agent.feeMin != null ? String(agent.feeMin) : '',
   245→    feeMax: agent.feeMax != null ? String(agent.feeMax) : '',
   246→    coins: String(agent.coins),
   247→    grantCoins: '',
   248→    style: p.style ?? '',
   249→    activeFrom: String(p.activeFrom ?? 5),
   250→    activeTo: String(p.activeTo ?? 17),
   251→    minDelaySec: String(p.minDelaySec ?? 20),
   252→    maxDelaySec: String(p.maxDelaySec ?? 90),
   253→    mergeWindowSec: String(p.mergeWindowSec ?? 7),
   254→    replyChance: String(p.replyChance ?? 0.9),
   255→    declineLines: (p.declineChances ?? []).join('\n'),
   256→  }
   257→}

   259→/**
   260→ * Shared dialog for creating and editing an AI agent.
   261→ * Create → POST with persona fields FLAT at top level + `coins`.
   262→ * Edit → PATCH with persona nested under `persona` + optional `grantCoins`.
   263→ */
   264→function AgentDialog({
   265→  open,
   266→  onOpenChange,
   267→  agent,
   268→  onSaved,
   269→}: {
   270→  open: boolean
   271→  onOpenChange: (open: boolean) => void
   272→  /** null = create mode */
   273→  agent: AiAgentDTO | null
   274→  onSaved: () => void
   275→}) {
   276→  const isEdit = !!agent
   277→  const [busy, setBusy] = useState(false)
   278→  const [uploading, setUploading] = useState(false)
   279→  const [f, setF] = useState<AgentFormState>(() => buildForm(agent))
   280→  const fileRef = useRef<HTMLInputElement>(null)
   281→  const isTeacher = f.role === 'TEACHER'
   282→
   283→  function set<K extends keyof AgentFormState>(key: K, value: AgentFormState[K]) {
   284→    setF((prev) => ({ ...prev, [key]: value }))
   285→  }
   286→
   287→  async function onPickAvatar(e: React.ChangeEvent<HTMLInputElement>) {
   288→    const file = e.target.files?.[0]
   289→    e.target.value = '' // allow re-picking the same file
   290→    if (!file) return
   291→    setUploading(true)
   292→    try {
   293→      set('avatar', (await api.uploadImage('avatars', file)).url)
   294→    } catch (err) {
   295→      toast.error('Could not process image', { description: errorMessage(err) })
   296→    } finally {
   297→      setUploading(false)
   298→    }
   299→  }
   300→
   301→  async function save() {
   302→    if (!f.name.trim()) {
   303→      toast.error('Name is required', { description: 'Give the agent a realistic human name.' })
   304→      return
   305→    }
   306→    setBusy(true)
   307→    try {
   308→      const minDelay = clampNum(f.minDelaySec, 20, 3, 3600)
   309→      const maxDelay = Math.max(minDelay, clampNum(f.maxDelaySec, 90, 5, 7200))
   310→      const declineChances = f.declineLines
   311→        .split('\n')
   312→        .map((s) => s.trim())
   313→        .filter(Boolean)
   314→        .slice(0, 6)
   315→
   316→      if (isEdit && agent) {
   317→        const grant = Math.max(0, Math.round(Number(f.grantCoins) || 0))
   318→        await api.adminUpdateAiAgent(agent.id, {
   319→          name: f.name.trim(),
   320→          ...(f.avatar && f.avatar !== agent.avatar ? { avatar: f.avatar } : {}),
   321→          headline: f.headline.trim(),
   322→          bio: f.bio.trim(),
   323→          city: f.city.trim(),
   324→          ...(isTeacher
   325→            ? {
   326→                subjects: f.subjects.trim(),
   327→                feeMin: Math.max(0, Number(f.feeMin) || 0),
   328→                feeMax: Math.max(0, Number(f.feeMax) || 0),
   329→              }
   330→            : {}),
   331→          ...(grant > 0 ? { grantCoins: grant } : {}),
   332→          persona: {
   333→            tagline: personaOf(agent.aiPersona).tagline?.trim() || f.headline.trim(),
   334→            style: f.style.trim(),
   335→            activeFrom: clampNum(f.activeFrom, 5, 0, 23),
   336→            activeTo: clampNum(f.activeTo, 17, 0, 24),
   337→            minDelaySec: minDelay,
   338→            maxDelaySec: maxDelay,
   339→            mergeWindowSec: clampNum(f.mergeWindowSec, 7, 2, 120),
   340→            replyChance: clampNum(f.replyChance, 0.9, 0, 1),
   341→            declineChances,
   342→          },
   343→        })
   344→        toast.success('Agent updated', {
   345→          description:
   346→            grant > 0
   347→              ? `${grant} coins granted to ${f.name.trim()}.`
   348→              : `${f.name.trim()}'s profile and persona were saved.`,
   349→        })
   350→      } else {
   351→        await api.adminCreateAiAgent({
   352→          name: f.name.trim(),
   353→          role: f.role,
   354→          ...(f.avatar ? { avatar: f.avatar } : {}),
   355→          headline: f.headline.trim(),
   356→          bio: f.bio.trim(),
   357→          city: f.city.trim(),
   358→          ...(isTeacher
   359→            ? {
   360→                subjects: f.subjects.trim(),
   361→                feeMin: Math.max(0, Number(f.feeMin) || 0),
   362→                feeMax: Math.max(0, Number(f.feeMax) || 0),
   363→                coins: Math.max(0, Math.round(Number(f.coins) || 0)),
   364→              }
   365→            : {}),
   366→          tagline: f.headline.trim() || (f.role === 'TEACHER' ? 'AI teacher on StudySir' : 'AI student on StudySir'),
   367→          style: f.style.trim(),
   368→          activeFrom: clampNum(f.activeFrom, 5, 0, 23),
   369→          activeTo: clampNum(f.activeTo, 17, 0, 24),
   370→          minDelaySec: minDelay,
   371→          maxDelaySec: maxDelay,
   372→          mergeWindowSec: clampNum(f.mergeWindowSec, 7, 2, 120),
   373→          replyChance: clampNum(f.replyChance, 0.9, 0, 1),
   374→          declineChances,
   375→        })
   376→        toast.success(`${f.name.trim()} is live on StudySir`, {
   377→          description: `AI ${f.role === 'TEACHER' ? 'teacher' : 'student'} created — it now chats with its own persona.`,
   378→        })
   379→      }
   380→      onSaved()
   381→    } catch (e) {
   382→      toast.error(isEdit ? 'Could not update agent' : 'Could not create agent', { description: errorMessage(e) })
   383→    } finally {
   384→      setBusy(false)
   385→    }
   386→  }
   387→
   388→  return (
   389→    <Dialog open={open} onOpenChange={onOpenChange}>
   390→      <DialogContent className="max-h-[90dvh] overflow-y-auto sm:max-w-lg">
   391→        <DialogHeader>
   392→          <DialogTitle className="flex items-center gap-2">
   393→            <Bot className="size-4 text-[#1877F2]" />
   394→            {isEdit ? `Edit ${agent?.name}` : 'New AI Agent'}
   395→          </DialogTitle>
   396→          <DialogDescription>
   397→            {isEdit
   398→              ? 'Profile, coins and humanlike persona for this agent.'
   399→              : 'Creates a humanlike user that chats, accepts and hires on StudySir.'}
   400→          </DialogDescription>
   401→        </DialogHeader>
   402→
   403→        <div className="space-y-4">
   404→          {/* role (create only — role is fixed after creation) */}
   405→          {!isEdit ? (
   406→            <div className="grid grid-cols-2 gap-2" role="radiogroup" aria-label="Agent role">
   407→              <button
   408→                type="button"
   409→                role="radio"
   410→                aria-checked={f.role === 'TEACHER'}
   411→                onClick={() => set('role', 'TEACHER')}
   412→                className={cn(
   413→                  'flex items-center justify-center gap-2 rounded-lg border px-3 py-2.5 text-sm font-semibold transition-colors',
   414→                  f.role === 'TEACHER'
   415→                    ? 'border-amber-500/60 bg-amber-500/10 text-amber-700 dark:text-amber-300'
   416→                    : 'border-border bg-muted/50 text-muted-foreground hover:bg-secondary'
   417→                )}
   418→              >
   419→                <GraduationCap className="size-4" />
   420→                AI Teacher
   421→              </button>
   422→              <button
   423→                type="button"
   424→                role="radio"
   425→                aria-checked={f.role === 'STUDENT'}
   426→                onClick={() => set('role', 'STUDENT')}
   427→                className={cn(
   428→                  'flex items-center justify-center gap-2 rounded-lg border px-3 py-2.5 text-sm font-semibold transition-colors',
   429→                  f.role === 'STUDENT'
   430→                    ? 'border-[#1877F2]/60 bg-[#1877F2]/10 text-[#1877F2] dark:text-blue-300'
   431→                    : 'border-border bg-muted/50 text-muted-foreground hover:bg-secondary'
   432→                )}
   433→              >
   434→                <BookUser className="size-4" />
   435→                AI Student
   436→              </button>
   437→            </div>
   438→          ) : null}
   439→
   440→          {/* name + avatar upload */}
   441→          <div className="flex items-end gap-3 rounded-lg bg-muted/50 p-3">
   442→            {f.avatar ? (
   443→              <img src={f.avatar} alt={`${f.name || 'Agent'} preview`} className="size-14 shrink-0 rounded-full object-cover" />
   444→            ) : (
   445→              <UserAvatar name={f.name || 'New Agent'} className="size-14" />
   446→            )}
   447→            <div className="min-w-0 flex-1 space-y-1.5">
   448→              <Label htmlFor="agent-name">Name *</Label>
   449→              <Input
   450→                id="agent-name"
   451→                value={f.name}
   452→                onChange={(e) => set('name', e.target.value)}
   453→                placeholder="e.g. Ayesha Khan"
   454→                maxLength={80}
   455→              />
   456→            </div>
   457→            <div className="shrink-0 pb-0.5">
   458→              <input
   459→                ref={fileRef}
   460→                type="file"
   461→                accept="image/*"
   462→                className="hidden"
   463→                onChange={(e) => void onPickAvatar(e)}

   465→                tabIndex={-1}

   472→          </div>
   473→
   474→          <div className="grid grid-cols-2 gap-3">

   489→                value={f.city}
   490→                onChange={(e) => set('city', e.target.value)}
   491→                placeholder="e.g. Lahore"
   492→                maxLength={80}
   493→              />

   495→          </div>

   497→          <div className="space-y-1.5">

   509→          {isTeacher ? (
